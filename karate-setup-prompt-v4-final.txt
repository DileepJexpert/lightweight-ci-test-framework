STEP 4 — WRITE karate-config.js
────────────────────────────────────────────────────────────────────────────────
Create src/test/resources/karate-config.js:

  function fn() {
    var env     = karate.env || 'qa';
    var baseUrl = karate.properties['service.base-url']
                  || java.lang.System.getenv('SERVICE_BASE_URL') || '';

    // baseUrl is ALWAYS required — you cannot hit any endpoint without it.
    if (baseUrl === '') {
      throw 'service.base-url is required. Use -Dservice.base-url=https://your-app.qa.internal';
    }

    karate.configure('connectTimeout', 5000);
    karate.configure('readTimeout',    15000);
    karate.configure('retry', { count: 10, interval: 250 });
    karate.configure('logPrettyRequest',  true);
    karate.configure('logPrettyResponse', true);

    var config = {
      env:       env,
      baseUrl:   baseUrl,
      authToken: null,
      defaultHeaders: {
        'Content-Type':     'application/json',
        'x-correlation-id': 'test-' + java.util.UUID.randomUUID()
      }
    };

    // ── STS credentials are OPTIONAL ───────────────────────────────────────────
    // If supplied  → fetch ONE Bearer token for the whole suite (cached by
    //                Karate) and inject it on every request.
    // If NOT supplied → run UNAUTHENTICATED. The @prod-safe @health and
    //                @performance read-only checks still pass; secured endpoints
    //                correctly return 401. This lets a quick health gate run
    //                without provisioning STS creds, instead of failing every
    //                scenario at config load.
    var stsTokenUrl  = karate.properties['sts.token.url']
                       || java.lang.System.getenv('STS_TOKEN_URL') || '';
    var clientId     = karate.properties['sts.client.id']
                       || java.lang.System.getenv('STS_CLIENT_ID') || '';
    var clientSecret = karate.properties['sts.client.secret']
                       || java.lang.System.getenv('STS_CLIENT_SECRET') || '';
    var scope        = karate.properties['sts.scope']
                       || java.lang.System.getenv('STS_SCOPE')
                       || 'dev-portal-suite';

    if (stsTokenUrl !== '' && clientId !== '' && clientSecret !== '') {
      // ── Fetch token ONCE for entire suite — cached automatically by Karate ───
      var session = karate.callSingle(
        'classpath:karate/features/helpers/get-sts-token.feature',
        { stsTokenUrl: stsTokenUrl, clientId: clientId,
          clientSecret: clientSecret, scope: scope }
      );
      var bearerToken = session.token;
      config.authToken = bearerToken;
      karate.log('[karate-config] Token acquired. env=' + env + ' baseUrl=' + baseUrl);

      // Inject Authorization + x-request-id on EVERY request
      karate.configure('headers', function() {
        return {
          'x-request-id':  'auto-' + java.util.UUID.randomUUID(),
          'Authorization': 'Bearer ' + bearerToken
        };
      });
    } else {
      karate.log('[karate-config] No STS credentials supplied — running UNAUTHENTICATED. '
        + 'Only @prod-safe @health / @performance checks will pass; secured endpoints '
        + 'will return 401. Pass -Dsts.token.url -Dsts.client.id -Dsts.client.secret '
        + 'to run the full @smoke suite.');
      // Still tag every request so it is traceable in the gateway/app logs.
      karate.configure('headers', function() {
        return { 'x-request-id': 'auto-' + java.util.UUID.randomUUID() };
      });
    }

    return config;
  }

STEP 5 — WRITE get-sts-token.feature
────────────────────────────────────────────────────────────────────────────────
Create src/test/java/karate/features/helpers/get-sts-token.feature:

  This MUST match how your STS expects credentials. The IDFC EntAuth STS uses
  multipart/form-data with grant_type=client_credentials. Equivalent curl:
    curl --request POST \
      --url https://app.uat-entauth.idfcfirstbank.com/platform/oauth/oauth2/token \
      --header 'content-type: multipart/form-data' \
      --form client_id=<client-id> \
      --form client_secret=<client-secret> \
      --form scope=dev-portal-suite \
      --form grant_type=client_credentials

  @ignore
  Feature: Fetch Bearer token from STS server using client_credentials grant

    Scenario:
      Given url stsTokenUrl
      And multipart field client_id     = clientId
      And multipart field client_secret = clientSecret
      And multipart field scope         = scope
      And multipart field grant_type    = 'client_credentials'
      When method post
      Then status 200
      And match response.access_token == '#notnull'
      * def token = response.access_token
      * karate.log('[get-sts-token] Token acquired, expires in: ' + response.expires_in + 's')

  NOTE: 'multipart field' makes Karate send content-type: multipart/form-data
  automatically — do NOT also set a Content-Type header manually.

  ALTERNATE FORMATS (only if your STS differs from the curl above):
    - x-www-form-urlencoded body:
        And header Content-Type = 'application/x-www-form-urlencoded'
        And request 'grant_type=client_credentials&client_id=' + clientId +
                    '&client_secret=' + clientSecret + '&scope=' + scope
    - Basic auth header + form body:
        And header Authorization = 'Basic ' + karate.toBase64(clientId + ':' + clientSecret)
        And request 'grant_type=client_credentials&scope=' + scope

  Performance checks:           @smoke @prod-safe @performance

  - * def correlationId = 'test-' + java.util.UUID.randomUUID()  for every scenario
